# Planets & Nations (Fabric, Minecraft 1.21.11)

Server-side mod: players join with a **vanilla client**. It adds four bounded planets, player nations with chunk claims, a planet travel menu, planet resources and craftable guns.

## Build

Needs Java 21.

```
gradle wrapper        # once, creates ./gradlew
./gradlew build       # jar ends up in build/libs/planets-1.0.0.jar
```

Put the jar and Fabric API (0.140.0+1.21.11) in the server's `mods` folder. If Gradle rejects the Loom or loader version, copy the current ones for 1.21.11 from https://fabricmc.net/develop/ into `gradle.properties`.

This project could not be compiled where it was written, so expect to fix a few Minecraft API name mismatches on the first build (the source uses Mojang mappings).

## Planets

| Planet | Border | Terrain |
|---|---|---|
| Overworld | 8000 x 8000 | vanilla |
| Frutiger Aero | 4000 x 4000 | bright meadows, floating glass water bubbles |
| Dreamcore | 4000 x 4000 | flat sunflower plains, ball pits, block towers, swing sets, retro TVs, toy chests (items from 2000-2016) |
| Autumn | 6000 x 6000 | tall birch forest, leaf litter and dead bushes, abandoned cabin settlements |

Borders are centred on 0,0. Sizes live in `Planet.java`. The dimensions are defined in `data/planets/dimension/*.json`.

## Travel

Everyone gets a **Planet Compass**; right-click it to open the planet menu. Also: `/planets`, `/compass` (get a new one). Nation members get a "Nation home" button.

## Nations

`/nation create <name>`, `claim`, `unclaim`, `invite <player>`, `accept`, `leave`, `kick <player>`, `disband`, `sethome`, `home`, `info [name]`, `list`, `here`, `map`.

A nation can claim 20 chunks plus 10 per extra member. Non-members cannot break, place or interact inside claimed chunks. Data is stored in `config/planets/nations.json`.

## Resources and guns

Planet resources are bonus drops from ores:

- Overworld: Saltpeter (coal)
- Frutiger Aero: Aero Crystal (lapis), Gloss Gel (emerald)
- Dreamcore: Plastic Scrap (copper), Battery Cell (gold)
- Autumn: Rusted Scrap (iron), Amber Resin (coal)

Right-click a **smithing table** to open the gunsmith (sneak + right-click for the normal table). Guns: Revolver, Hunting Rifle, Aero Beam Rifle, Foam Blaster, Scrap Shotgun. Each needs its own ammo, so players are pushed to trade between planets. Stats are in `Gun.java`, costs in `Gunsmith.java`.

## Known limits

- Explosions (TNT, creepers) can still damage claims; add a mixin if you want that blocked.
- Server operators have no claim bypass yet.
- The planets use the vanilla Overworld dimension type, so there is no custom sky or fixed time. Biome choice carries the mood.
- The vanilla world border is square, so each planet is one size x size square.

package com.planets;

import net.fabricmc.fabric.api.event.player.PlayerBlockBreakEvents;
import net.fabricmc.fabric.api.event.player.UseBlockCallback;
import net.fabricmc.fabric.api.event.player.UseItemCallback;
import net.minecraft.core.BlockPos;
import net.minecraft.server.level.ServerLevel;
import net.minecraft.server.level.ServerPlayer;
import net.minecraft.world.InteractionResult;
import net.minecraft.world.item.ItemStack;
import net.minecraft.world.level.block.Block;
import net.minecraft.world.level.block.Blocks;
import net.minecraft.world.level.block.state.BlockState;

/** Land protection, bonus ore drops, gunsmith table and item right-clicks. */
public final class BlockEvents {
    private BlockEvents() {}

    public static void register() {
        // Protection: breaking
        PlayerBlockBreakEvents.BEFORE.register((level, player, pos, state, blockEntity) -> {
            if (player instanceof ServerPlayer sp && level instanceof ServerLevel sl
                    && !PlanetsMod.NATIONS.canBuild(sp, sl, pos)) {
                PlanetsMod.NATIONS.deny(sp, sl, pos);
                return false;
            }
            return true;
        });

        // Planet resources: bonus drops
        PlayerBlockBreakEvents.AFTER.register((level, player, pos, state, blockEntity) -> {
            if (!(player instanceof ServerPlayer sp) || !(level instanceof ServerLevel sl) || sp.isCreative()) return;
            Planet planet = Planet.of(sl);
            if (planet == null) return;
            ItemStack extra = PlanetResources.extraDrop(planet, state, sl.getRandom());
            if (extra != null) {
                Block.popResource(sl, pos, extra);
            }
        });

        // Right-click on a block: guns, compass, gunsmith table, then protection
        UseBlockCallback.EVENT.register((player, level, hand, hit) -> {
            if (!(player instanceof ServerPlayer sp) || !(level instanceof ServerLevel sl)) {
                return InteractionResult.PASS;
            }
            if (ItemUse.handle(sp, sp.getItemInHand(hand))) {
                return InteractionResult.SUCCESS;
            }

            BlockPos clicked = hit.getBlockPos();
            BlockState state = sl.getBlockState(clicked);
            if (state.is(Blocks.SMITHING_TABLE) && !sp.isShiftKeyDown()) {
                Gunsmith.open(sp);
                return InteractionResult.SUCCESS;
            }

            BlockPos placed = clicked.relative(hit.getDirection());
            if (!PlanetsMod.NATIONS.canBuild(sp, sl, clicked)) {
                PlanetsMod.NATIONS.deny(sp, sl, clicked);
                return InteractionResult.FAIL;
            }
            if (!PlanetsMod.NATIONS.canBuild(sp, sl, placed)) {
                PlanetsMod.NATIONS.deny(sp, sl, placed);
                return InteractionResult.FAIL;
            }
            return InteractionResult.PASS;
        });

        // Right-click in the air
        UseItemCallback.EVENT.register((player, level, hand) -> {
            if (player instanceof ServerPlayer sp && ItemUse.handle(sp, sp.getItemInHand(hand))) {
                return InteractionResult.SUCCESS;
            }
            return InteractionResult.PASS;
        });
    }
}

package com.planets;

import java.util.function.BiConsumer;
import net.minecraft.server.level.ServerPlayer;
import net.minecraft.world.Container;
import net.minecraft.world.entity.player.Inventory;
import net.minecraft.world.entity.player.Player;
import net.minecraft.world.inventory.ChestMenu;
import net.minecraft.world.inventory.ClickType;
import net.minecraft.world.inventory.MenuType;
import net.minecraft.world.item.ItemStack;

/** A chest-style GUI whose items are buttons. Items can never be taken out. */
public class ButtonMenu extends ChestMenu {
    private final int rows;
    private final BiConsumer<ServerPlayer, Integer> handler;

    public ButtonMenu(MenuType<?> type, int syncId, Inventory inv, Container container, int rows,
                      BiConsumer<ServerPlayer, Integer> handler) {
        super(type, syncId, inv, container, rows);
        this.rows = rows;
        this.handler = handler;
    }

    @Override
    public void clicked(int slot, int button, ClickType type, Player player) {
        if (slot >= 0 && slot < rows * 9 && type == ClickType.PICKUP && player instanceof ServerPlayer sp) {
            handler.accept(sp, slot);
        }
        // Never call super: nothing may be moved. Resync so the client forgets any drag.
        if (player.containerMenu == this) {
            sendAllDataToRemote();
        }
    }

    @Override
    public ItemStack quickMoveStack(Player player, int index) {
        return ItemStack.EMPTY;
    }

    @Override
    public boolean stillValid(Player player) {
        return true;
    }

    public static MenuType<?> typeFor(int rows) {
        return switch (rows) {
            case 1 -> MenuType.GENERIC_9x1;
            case 2 -> MenuType.GENERIC_9x2;
            case 3 -> MenuType.GENERIC_9x3;
            case 4 -> MenuType.GENERIC_9x4;
            case 5 -> MenuType.GENERIC_9x5;
            default -> MenuType.GENERIC_9x6;
        };
    }
}

package com.planets;

import java.util.ArrayList;
import java.util.List;
import net.minecraft.ChatFormatting;
import net.minecraft.core.component.DataComponents;
import net.minecraft.network.chat.Component;
import net.minecraft.server.level.ServerPlayer;
import net.minecraft.sounds.SoundEvents;
import net.minecraft.sounds.SoundSource;
import net.minecraft.world.SimpleContainer;
import net.minecraft.world.SimpleMenuProvider;
import net.minecraft.world.entity.player.Inventory;
import net.minecraft.world.item.Item;
import net.minecraft.world.item.ItemStack;
import net.minecraft.world.item.Items;
import net.minecraft.world.item.component.ItemLore;

/**
 * The gunsmith: right-click a smithing table (not sneaking) to craft ammo, guns and the compass.
 * Done in code because vanilla recipes cannot require items with custom names.
 */
public final class Gunsmith {
    private Gunsmith() {}

    private record Req(Item vanilla, ModItems.Id custom, int n) {
        static Req v(Item item, int n) { return new Req(item, null, n); }
        static Req c(ModItems.Id id, int n) { return new Req(null, id, n); }

        Component label() {
            return custom != null ? Component.literal(custom.display) : new ItemStack(vanilla).getHoverName();
        }
    }

    private record Recipe(ItemStack result, List<Req> cost) {}

    private static List<Recipe> recipes;

    private static List<Recipe> recipes() {
        if (recipes == null) {
            List<Recipe> r = new ArrayList<>();
            // Ammo
            r.add(new Recipe(ModItems.make(ModItems.Id.LIGHT_ROUNDS, 16),
                    List.of(Req.v(Items.IRON_INGOT, 1), Req.v(Items.GUNPOWDER, 1), Req.c(ModItems.Id.SALTPETER, 1))));
            r.add(new Recipe(ModItems.make(ModItems.Id.RIFLE_ROUNDS, 8),
                    List.of(Req.v(Items.IRON_INGOT, 2), Req.v(Items.GUNPOWDER, 1), Req.c(ModItems.Id.SALTPETER, 2))));
            r.add(new Recipe(ModItems.make(ModItems.Id.AERO_CELLS, 12),
                    List.of(Req.c(ModItems.Id.AERO_CRYSTAL, 2), Req.c(ModItems.Id.GLOSS_GEL, 1))));
            r.add(new Recipe(ModItems.make(ModItems.Id.FOAM_DARTS, 24),
                    List.of(Req.c(ModItems.Id.PLASTIC_SCRAP, 3))));
            r.add(new Recipe(ModItems.make(ModItems.Id.SCRAP_SHELLS, 8),
                    List.of(Req.c(ModItems.Id.RUSTED_SCRAP, 4), Req.v(Items.GUNPOWDER, 1), Req.c(ModItems.Id.AMBER_RESIN, 1))));
            // Guns
            r.add(new Recipe(ModItems.make(ModItems.Id.REVOLVER, 1),
                    List.of(Req.v(Items.IRON_INGOT, 8), Req.v(Items.COPPER_INGOT, 2), Req.v(Items.FLINT, 1))));
            r.add(new Recipe(ModItems.make(ModItems.Id.HUNTING_RIFLE, 1),
                    List.of(Req.v(Items.IRON_INGOT, 12), Req.v(Items.STICK, 6), Req.v(Items.DIAMOND, 1))));
            r.add(new Recipe(ModItems.make(ModItems.Id.AERO_BEAM_RIFLE, 1),
                    List.of(Req.c(ModItems.Id.AERO_CRYSTAL, 6), Req.c(ModItems.Id.GLOSS_GEL, 4),
                            Req.v(Items.IRON_INGOT, 6), Req.v(Items.DIAMOND, 1))));
            r.add(new Recipe(ModItems.make(ModItems.Id.FOAM_BLASTER, 1),
                    List.of(Req.c(ModItems.Id.PLASTIC_SCRAP, 8), Req.c(ModItems.Id.BATTERY_CELL, 3), Req.v(Items.IRON_INGOT, 2))));
            r.add(new Recipe(ModItems.make(ModItems.Id.SCRAP_SHOTGUN, 1),
                    List.of(Req.c(ModItems.Id.RUSTED_SCRAP, 16), Req.c(ModItems.Id.AMBER_RESIN, 3), Req.v(Items.IRON_INGOT, 4))));
            // Travel
            r.add(new Recipe(ModItems.make(ModItems.Id.PLANET_COMPASS, 1),
                    List.of(Req.v(Items.IRON_INGOT, 1), Req.v(Items.REDSTONE, 1))));
            recipes = r;
        }
        return recipes;
    }

    private static final int FIRST_SLOT = 9;
    private static final int INFO_SLOT = 31;

    public static void open(ServerPlayer player) {
        SimpleContainer container = new SimpleContainer(36);
        render(player, container);
        player.openMenu(new SimpleMenuProvider(
                (syncId, inv, p) -> new ButtonMenu(ButtonMenu.typeFor(4), syncId, inv, container, 4,
                        (sp, slot) -> onClick(sp, slot, container)),
                Component.literal("Gunsmith")));
    }

    private static void onClick(ServerPlayer player, int slot, SimpleContainer container) {
        int index = slot - FIRST_SLOT;
        List<Recipe> list = recipes();
        if (index < 0 || index >= list.size()) return;
        Recipe recipe = list.get(index);

        for (Req req : recipe.cost()) {
            if (count(player, req) < req.n()) {
                player.displayClientMessage(Component.literal("You are missing materials.").withStyle(ChatFormatting.RED), true);
                return;
            }
        }
        for (Req req : recipe.cost()) take(player, req);
        player.getInventory().placeItemBackInInventory(recipe.result().copy());
        player.level().playSound(null, player.blockPosition(), SoundEvents.ANVIL_USE, SoundSource.PLAYERS, 0.8f, 1.2f);
        render(player, container);
    }

    private static void render(ServerPlayer player, SimpleContainer container) {
        ItemStack filler = new ItemStack(Items.GRAY_STAINED_GLASS_PANE);
        filler.set(DataComponents.CUSTOM_NAME, Component.literal(" "));
        for (int i = 0; i < container.getContainerSize(); i++) container.setItem(i, filler.copy());

        List<Recipe> list = recipes();
        for (int i = 0; i < list.size(); i++) {
            Recipe recipe = list.get(i);
            ItemStack shown = recipe.result().copy();
            List<Component> lore = new ArrayList<>();
            ItemLore existing = shown.get(DataComponents.LORE);
            if (existing != null) lore.addAll(existing.lines());
            lore.add(ModItems.text("Cost:", ChatFormatting.WHITE));
            boolean canCraft = true;
            for (Req req : recipe.cost()) {
                int have = count(player, req);
                boolean enough = have >= req.n();
                if (!enough) canCraft = false;
                lore.add(Component.empty()
                        .append(ModItems.text((enough ? "  ✔ " : "  ✘ ") + Math.min(have, 999) + "/" + req.n() + " ",
                                enough ? ChatFormatting.GREEN : ChatFormatting.RED))
                        .append(req.label().copy().withStyle(enough ? ChatFormatting.GREEN : ChatFormatting.RED)));
            }
            lore.add(ModItems.text(canCraft ? "Click to craft" : "Missing materials",
                    canCraft ? ChatFormatting.GREEN : ChatFormatting.RED));
            shown.set(DataComponents.LORE, new ItemLore(lore));
            container.setItem(FIRST_SLOT + i, shown);
        }

        ItemStack info = new ItemStack(Items.SMITHING_TABLE);
        info.set(DataComponents.CUSTOM_NAME, ModItems.text("Gunsmith", ChatFormatting.GOLD));
        info.set(DataComponents.LORE, new ItemLore(List.of(
                ModItems.text("Planet resources are bonus drops from ores:", ChatFormatting.GRAY),
                ModItems.text("Overworld: Saltpeter  (coal)", ChatFormatting.GRAY),
                ModItems.text("Frutiger Aero: Aero Crystal, Gloss Gel  (lapis, emerald)", ChatFormatting.GRAY),
                ModItems.text("Dreamcore: Plastic Scrap, Battery Cell  (copper, gold)", ChatFormatting.GRAY),
                ModItems.text("Autumn: Rusted Scrap, Amber Resin  (iron, coal)", ChatFormatting.GRAY),
                ModItems.text("Sneak + right-click the table to use it normally.", ChatFormatting.DARK_GRAY))));
        container.setItem(INFO_SLOT, info);
    }

    private static boolean matches(ItemStack s, Req req) {
        if (s.isEmpty()) return false;
        if (req.custom() != null) return ModItems.idOf(s) == req.custom();
        return !ModItems.isModItem(s) && s.is(req.vanilla());
    }

    private static int count(ServerPlayer p, Req req) {
        int total = 0;
        Inventory inv = p.getInventory();
        for (int i = 0; i < inv.getContainerSize(); i++) {
            ItemStack s = inv.getItem(i);
            if (matches(s, req)) total += s.getCount();
        }
        return total;
    }

    private static void take(ServerPlayer p, Req req) {
        int left = req.n();
        Inventory inv = p.getInventory();
        for (int i = 0; i < inv.getContainerSize() && left > 0; i++) {
            ItemStack s = inv.getItem(i);
            if (matches(s, req)) {
                int t = Math.min(left, s.getCount());
                s.shrink(t);
                left -= t;
            }
        }
    }
}

package com.planets;

import net.minecraft.ChatFormatting;
import net.minecraft.network.chat.Component;
import net.minecraft.server.level.ServerLevel;
import net.minecraft.server.level.ServerPlayer;
import net.minecraft.sounds.SoundEvents;
import net.minecraft.sounds.SoundSource;
import net.minecraft.util.RandomSource;
import net.minecraft.world.entity.Entity;
import net.minecraft.world.entity.LivingEntity;
import net.minecraft.world.entity.projectile.ProjectileUtil;
import net.minecraft.world.entity.player.Inventory;
import net.minecraft.world.item.ItemStack;
import net.minecraft.world.level.ClipContext;
import net.minecraft.world.phys.AABB;
import net.minecraft.world.phys.BlockHitResult;
import net.minecraft.world.phys.EntityHitResult;
import net.minecraft.world.phys.HitResult;
import net.minecraft.world.phys.Vec3;

/** Right-click behaviour of the Planet Compass and the guns. */
public final class ItemUse {
    private ItemUse() {}

    /** Returns true if the stack was one of ours and the click has been handled. */
    public static boolean handle(ServerPlayer player, ItemStack stack) {
        ModItems.Id id = ModItems.idOf(stack);
        if (id == null) return false;
        if (id == ModItems.Id.PLANET_COMPASS) {
            PlanetTravel.openMenu(player);
            return true;
        }
        Gun gun = Gun.of(id);
        if (gun == null) return false;
        fire(player, stack, gun);
        return true;
    }

    private static void fire(ServerPlayer p, ItemStack stack, Gun gun) {
        ServerLevel level = (ServerLevel) p.level();
        if (p.getCooldowns().isOnCooldown(stack)) return;

        if (!consumeAmmo(p, gun.ammo)) {
            p.displayClientMessage(Component.literal("Out of " + gun.ammo.display + "!").withStyle(ChatFormatting.RED), true);
            level.playSound(null, p.blockPosition(), SoundEvents.DISPENSER_FAIL, SoundSource.PLAYERS, 1.0f, 1.2f);
            p.getCooldowns().addCooldown(stack, 8);
            return;
        }
        p.getCooldowns().addCooldown(stack, gun.cooldown);
        level.playSound(null, p.blockPosition(), gun.sound, SoundSource.PLAYERS, 1.2f, gun.pitch);

        for (int i = 0; i < gun.pellets; i++) {
            shoot(level, p, gun);
        }
    }

    private static void shoot(ServerLevel level, ServerPlayer p, Gun gun) {
        Vec3 eye = p.getEyePosition();
        Vec3 dir = p.getLookAngle();
        if (gun.spread > 0) {
            RandomSource r = p.getRandom();
            dir = dir.add((r.nextDouble() - 0.5) * 2 * gun.spread,
                    (r.nextDouble() - 0.5) * 2 * gun.spread,
                    (r.nextDouble() - 0.5) * 2 * gun.spread).normalize();
        }

        Vec3 end = eye.add(dir.scale(gun.range));
        BlockHitResult blockHit = level.clip(new ClipContext(eye, end, ClipContext.Block.COLLIDER, ClipContext.Fluid.NONE, p));
        if (blockHit.getType() != HitResult.Type.MISS) {
            end = blockHit.getLocation();
        }

        AABB box = new AABB(eye, end).inflate(1.0);
        EntityHitResult entityHit = ProjectileUtil.getEntityHitResult(p, eye, end, box,
                e -> !e.isSpectator() && e.isPickable() && e != p, eye.distanceToSqr(end));
        if (entityHit != null) {
            end = entityHit.getLocation();
            Entity target = entityHit.getEntity();
            if (target instanceof LivingEntity living) {
                boolean friendly = target instanceof ServerPlayer tp
                        && PlanetsMod.NATIONS.sameNation(p.getUUID(), tp.getUUID());
                if (!friendly) {
                    living.hurtServer(level, level.damageSources().playerAttack(p), gun.damage);
                    if (gun.knockback > 0) {
                        living.knockback(gun.knockback, -dir.x, -dir.z);
                    }
                }
            }
        }

        double dist = eye.distanceTo(end);
        for (double d = 1.5; d < dist; d += 1.2) {
            Vec3 pt = eye.add(dir.scale(d));
            level.sendParticles(gun.particle, pt.x, pt.y, pt.z, 1, 0, 0, 0, 0);
        }
    }

    private static boolean consumeAmmo(ServerPlayer p, ModItems.Id ammo) {
        if (p.isCreative()) return true;
        Inventory inv = p.getInventory();
        for (int i = 0; i < inv.getContainerSize(); i++) {
            ItemStack s = inv.getItem(i);
            if (ModItems.idOf(s) == ammo) {
                s.shrink(1);
                return true;
            }
        }
        return false;
    }

package com.planets;

import com.google.gson.Gson;
import com.google.gson.GsonBuilder;
import java.nio.file.Files;

import java.util.HashMap;

import net.fabricmc.loader.api.FabricLoader;
import net.minecraft.ChatFormatting;

import net.minecraft.server.level.ServerLevel;
import net.minecraft.server.level.ServerPlayer;

/** Nations, members and chunk claims. Saved to config/planets/nations.json. */
public class NationManager {
    public static final int BASE_CLAIMS = 20;
    public static final int CLAIMS_PER_EXTRA_MEMBER = 10;

    public static class Nation {
        public String name;
        public String owner;
        public Set<String> members = new LinkedHashSet<>();
        public Set<String> claims = new LinkedHashSet<>();
        public String homePlanet;
        public double hx;
        public double hy;
        public double hz;
    }

    private static class Data {
        Map<String, Nation> nations = new LinkedHashMap<>();
        Map<String, String> names = new HashMap<>(); // uuid -> last known name
    }

    private static final Gson GSON = new GsonBuilder().setPrettyPrinting().create();

    private Data data = new Data();
    private final Map<String, String> claimIndex = new HashMap<>();  // claim key -> nation key
    private final Map<String, String> memberIndex = new HashMap<>(); // uuid -> nation key
    public final Map<UUID, String> invites = new HashMap<>();        // invited player -> nation key
    private Path file;

    public void load() {
        file = FabricLoader.getInstance().getConfigDir().resolve("planets/nations.json");
        try {
            if (Files.exists(file)) {
                Data loaded = GSON.fromJson(Files.readString(file), Data.class);
                if (loaded != null) data = loaded;
            }
        } catch (Exception e) {
            PlanetsMod.LOGGER.error("Could not read nations.json", e);
        }
        reindex();
    }

    public void save() {
        try {
            Files.createDirectories(file.getParent());
            Files.writeString(file, GSON.toJson(data));
        } catch (Exception e) {
            PlanetsMod.LOGGER.error("Could not write nations.json", e);
        }
    }

    private void reindex() {
        claimIndex.clear();
        memberIndex.clear();
        for (Map.Entry<String, Nation> e : data.nations.entrySet()) {
            for (String c : e.getValue().claims) claimIndex.put(c, e.getKey());
            for (String m : e.getValue().members) memberIndex.put(m, e.getKey());
        }
    }

    // ---- lookups ----

    public static String claimKey(String planetId, int cx, int cz) {
        return planetId + ":" + cx + ":" + cz;
    }

    public Nation byName(String name) {
        return data.nations.get(name.toLowerCase(Locale.ROOT));
    }

    public Nation byPlayer(UUID uuid) {
        String key = memberIndex.get(uuid.toString());
        return key == null ? null : data.nations.get(key);
    }

    public Nation claimAt(String planetId, int cx, int cz) {
        String key = claimIndex.get(claimKey(planetId, cx, cz));
        return key == null ? null : data.nations.get(key);
    }

    public Iterable<Nation> all() {
        return data.nations.values();
    }

    public boolean sameNation(UUID a, UUID b) {
        Nation n = byPlayer(a);
        return n != null && n.members.contains(b.toString());
    }

    public int maxClaims(Nation n) {
        return BASE_CLAIMS + CLAIMS_PER_EXTRA_MEMBER * Math.max(0, n.members.size() - 1);
    }

    public void remember(ServerPlayer p) {
        data.names.put(p.getUUID().toString(), p.getGameProfile().name());
        save();
    }

    public String nameOf(String uuid) {
        return data.names.getOrDefault(uuid, uuid.substring(0, 8));
    }

    public String uuidByName(String name) {
        for (Map.Entry<String, String> e : data.names.entrySet()) {
            if (e.getValue().equalsIgnoreCase(name)) return e.getKey();
        }
        return null;
    }

    // ---- changes ----

    public Nation create(String name, ServerPlayer owner) {
        Nation n = new Nation();
        n.name = name;
        n.owner = owner.getUUID().toString();
        n.members.add(n.owner);
        data.nations.put(name.toLowerCase(Locale.ROOT), n);
        reindex();
        save();
        return n;
    }

    public void disband(Nation n) {
        data.nations.remove(n.name.toLowerCase(Locale.ROOT));
        invites.values().removeIf(v -> v.equalsIgnoreCase(n.name));
        reindex();
        save();
    }

    public void addMember(Nation n, UUID uuid) {
        n.members.add(uuid.toString());
        reindex();
        save();
    }

    public void removeMember(Nation n, String uuid) {
        n.members.remove(uuid);
        reindex();
        save();
    }

    public void addClaim(Nation n, String key) {
        n.claims.add(key);
        claimIndex.put(key, n.name.toLowerCase(Locale.ROOT));
        save();
    }

    public void removeClaim(Nation n, String key) {
        n.claims.remove(key);
        claimIndex.remove(key);
        save();
    }

    // ---- protection ----

    public boolean canBuild(ServerPlayer p, ServerLevel level, BlockPos pos) {
        Planet planet = Planet.of(level);
        if (planet == null) return true;
        Nation n = claimAt(planet.id, pos.getX() >> 4, pos.getZ() >> 4);
        return n == null || n.members.contains(p.getUUID().toString());
    }

    public void deny(ServerPlayer p, ServerLevel level, BlockPos pos) {
        Planet planet = Planet.of(level);
        Nation n = planet == null ? null : claimAt(planet.id, pos.getX() >> 4, pos.getZ() >> 4);
        String who = n == null ? "another nation" : n.name;
        p.displayClientMessage(Component.literal("This land belongs to " + who + ".")
                .withStyle(ChatFormatting.RED), true);
    }

        this.icon = icon;
        this.color = color;
        this.tagline = tagline;
        this.resources = resources;
    }

    /** Which planet a level is, or null for the Nether, the End or anything else. */
    public static Planet of(ServerLevel level) {
        // Matching on the dimension key's text keeps this independent of
        // identifier class renames between Minecraft versions.
        String key = level.dimension().toString();
        if (key.contains("minecraft:overworld")) return OVERWORLD;
        for (Planet p : values()) {
            if (p != OVERWORLD && key.contains("planets:" + p.id)) return p;
        }
        return null;
    }

    public ServerLevel level(MinecraftServer server) {
        for (ServerLevel l : server.getAllLevels()) {
            if (of(l) == this) return l;
        }
        return null;
    }

    public static Planet byId(String id) {
        for (Planet p : values()) {
            if (p.id.equals(id)) return p;
        }
        return null;
    }

/**
 * Adds planet flavour to chunks the first time a player is near them: extra ores, Dreamcore toys and
 * playgrounds, Frutiger Aero glass bubbles, Autumn leaf litter and abandoned cabins.
 * Everything stays inside the chunk it is generated for, so it never touches unloaded chunks.
 * Chunks that were already decorated are remembered in config/planets/decorated.json.

public final class PlanetDecorator {
    private PlanetDecorator() {}

    private static final Gson GSON = new Gson();
    private static final Map<String, Set<Long>> DONE = new HashMap<>();
    private static Path file;

    public static void load() {
        file = FabricLoader.getInstance().getConfigDir().resolve("planets/decorated.json");
        try {
            if (Files.exists(file)) {
                Type type = new TypeToken<Map<String, Set<Long>>>() {}.getType();
                Map<String, Set<Long>> loaded = GSON.fromJson(Files.readString(file), type);
                if (loaded != null) DONE.putAll(loaded);
            }
        } catch (Exception e) {
            PlanetsMod.LOGGER.error("Could not read decorated.json", e);
        }
    }

    public static void save() {
        try {
            Files.createDirectories(file.getParent());
            Files.writeString(file, GSON.toJson(DONE));
        } catch (Exception e) {
            PlanetsMod.LOGGER.error("Could not write decorated.json", e);
        }
    }

    public static void tick(MinecraftServer server) {
        if (server.getTickCount() % 10 != 0) return;
        int budget = 3; // chunks per pass, keeps the server smooth
        for (ServerPlayer player : server.getPlayerList().getPlayers()) {
            if (budget <= 0) return;
            ServerLevel level = (ServerLevel) player.level();
            Planet planet = Planet.of(level);
            if (planet == null || planet == Planet.OVERWORLD) continue;
            Set<Long> done = DONE.computeIfAbsent(planet.id, k -> new HashSet<>());
            int pcx = player.blockPosition().getX() >> 4;
            int pcz = player.blockPosition().getZ() >> 4;
            for (int dx = -5; dx <= 5 && budget > 0; dx++) {
                for (int dz = -5; dz <= 5 && budget > 0; dz++) {
                    int cx = pcx + dx;
                    int cz = pcz + dz;
                    long key = ChunkPos.asLong(cx, cz);
                    if (done.contains(key)) continue;
                    LevelChunk chunk = level.getChunkSource().getChunkNow(cx, cz);
                    if (chunk == null) continue;
                    done.add(key);
                    decorate(level, planet, cx, cz);
                    budget--;
                }
            }
        }
    }

    private static void decorate(ServerLevel level, Planet planet, int cx, int cz) {
        RandomSource rng = RandomSource.create(level.getSeed() * 341873128712L
                + cx * 132897987541L + cz * 42317861L + planet.ordinal());
        switch (planet) {
            case FRUTIGER_AERO -> {
                ores(level, rng, cx, cz, new Block[] {Blocks.LAPIS_ORE, Blocks.EMERALD_ORE}, 5);
                if (rng.nextInt(14) == 0) bubble(level, rng, cx, cz);
            }
            case DREAMCORE -> dreamcore(level, rng, cx, cz);
            case AUTUMN -> autumn(level, rng, cx, cz);
            default -> { }
        }
    }

    private static void autumn(ServerLevel level, RandomSource rng, int cx, int cz) {
        int bx = cx * 16;
        int bz = cz * 16;

        // Dead leaves and bushes on the ground
        for (int x = 0; x < 16; x++) {
            for (int z = 0; z < 16; z++) {
                if (rng.nextFloat() > 0.4f) continue;
                int wx = bx + x;
                int wz = bz + z;
                int y = surface(level, wx, wz);
                BlockPos ground = new BlockPos(wx, y - 1, wz);
                BlockState below = level.getBlockState(ground);
                if (below.is(Blocks.GRASS_BLOCK) || below.is(Blocks.PODZOL) || below.is(Blocks.DIRT)) {
                    BlockPos above = ground.above();
                    if (level.getBlockState(above).isAir()) {
                        Block deco = rng.nextInt(12) == 0 ? Blocks.DEAD_BUSH : Blocks.LEAF_LITTER;
                        level.setBlock(above, deco.defaultBlockState(), 2);
                    }
                }
            }
        }

        // Settlements: some 6x6-chunk regions hold an old village around a centre chunk
        int rx = Math.floorDiv(cx, 6);
        int rz = Math.floorDiv(cz, 6);
        RandomSource region = RandomSource.create(level.getSeed() ^ (rx * 73856093L) ^ (rz * 19349663L));
        boolean village = region.nextInt(2) == 0;
        int centreX = rx * 6 + region.nextInt(6);
        int centreZ = rz * 6 + region.nextInt(6);
        double chance = (village && Math.abs(cx - centreX) <= 1 && Math.abs(cz - centreZ) <= 1) ? 0.7 : 0.012;
        if (rng.nextDouble() < chance) {
            cabin(level, rng, cx, cz);
        }
    }

    private static void cabin(ServerLevel level, RandomSource rng, int cx, int cz) {
        int ox = cx * 16 + 2 + rng.nextInt(6);
        int oz = cz * 16 + 2 + rng.nextInt(6);
        int y = surface(level, ox + 3, oz + 3);
        if (!level.getFluidState(new BlockPos(ox + 3, y - 1, oz + 3)).isEmpty()) return;

        for (int dx = 0; dx <= 6; dx++) {
            for (int dz = 0; dz <= 6; dz++) {
                int x = ox + dx;
                int z = oz + dz;
                // foundation
                for (int yy = y - 2; yy >= y - 5; yy--) {
                    BlockPos p = new BlockPos(x, yy, z);
                    BlockState s = level.getBlockState(p);
                    if (s.isAir() || s.canBeReplaced()) {
                        level.setBlock(p, Blocks.COBBLESTONE.defaultBlockState(), 2);
                    }
                }
                // floor
                set(level, x, y - 1, z, rng.nextInt(3) == 0 ? Blocks.MOSSY_COBBLESTONE : Blocks.SPRUCE_PLANKS);
                // clear the space above
                for (int dy = 0; dy <= 6; dy++) {
                    set(level, x, y + dy, z, Blocks.AIR);
                }
                boolean perimeter = dx == 0 || dx == 6 || dz == 0 || dz == 6;
                boolean corner = (dx == 0 || dx == 6) && (dz == 0 || dz == 6);
                if (perimeter) {
                    for (int dy = 0; dy <= 3; dy++) {
                        if (dx == 3 && dz == 0 && dy < 2) continue; // doorway
                        if (corner) {
                            set(level, x, y + dy, z, Blocks.SPRUCE_LOG);
                        } else if (rng.nextInt(4) == 0) {
                            if (rng.nextInt(2) == 0) set(level, x, y + dy, z, Blocks.COBWEB);
                        } else {
                            set(level, x, y + dy, z, rng.nextInt(5) == 0 ? Blocks.MOSSY_COBBLESTONE : Blocks.SPRUCE_PLANKS);
                        }
                    }
                } else if (rng.nextInt(5) < 2) {
                    set(level, x, y, z, Blocks.LEAF_LITTER);
                }
                // patchy roof
                if (rng.nextInt(10) < 6) {
                    set(level, x, y + 4, z, Blocks.SPRUCE_SLAB);
                }
            }
        }

        if (rng.nextInt(4) != 0) {
            BlockPos pos = new BlockPos(ox + 5, y, oz + 5);
            level.setBlock(pos, Blocks.BARREL.defaultBlockState(), 3);
            if (level.getBlockEntity(pos) instanceof Container chest) {
                int n = 2 + rng.nextInt(4);
                for (int i = 0; i < n; i++) {
                    int slot = rng.nextInt(chest.getContainerSize());
                    switch (rng.nextInt(5)) {
                        case 0 -> chest.setItem(slot, ModItems.make(ModItems.Id.RUSTED_SCRAP, 2 + rng.nextInt(5)));
                        case 1 -> chest.setItem(slot, ModItems.make(ModItems.Id.AMBER_RESIN, 1 + rng.nextInt(3)));
                        case 2 -> chest.setItem(slot, ModItems.make(ModItems.Id.SALTPETER, 1 + rng.nextInt(2)));
                        case 3 -> chest.setItem(slot, new net.minecraft.world.item.ItemStack(Items.IRON_INGOT, 1 + rng.nextInt(3)));
                        default -> chest.setItem(slot, new net.minecraft.world.item.ItemStack(Items.PUMPKIN_PIE, 1 + rng.nextInt(2)));
                    }
                }
            }
        }
    }
}

    }

    private static boolean isGold(BlockState s) {
        return s.is(Blocks.GOLD_ORE) || s.is(Blocks.DEEPSLATE_GOLD_ORE);
    }

    /** The bonus drop for breaking {@code state} on {@code planet}, or null. */
    public static ItemStack extraDrop(Planet planet, BlockState state, RandomSource r) {
        switch (planet) {
            case OVERWORLD:
                if (isCoal(state) && r.nextFloat() < 0.6f) {
                    return ModItems.make(ModItems.Id.SALTPETER, 1 + r.nextInt(2));
                }
                break;
            case FRUTIGER_AERO:
                if (isLapis(state)) return ModItems.make(ModItems.Id.AERO_CRYSTAL, 2 + r.nextInt(3));
                if (isEmerald(state)) return ModItems.make(ModItems.Id.GLOSS_GEL, 1 + r.nextInt(2));
                break;
            case DREAMCORE:
                if (isCopper(state)) return ModItems.make(ModItems.Id.PLASTIC_SCRAP, 2 + r.nextInt(3));
                if (isGold(state)) return ModItems.make(ModItems.Id.BATTERY_CELL, 1 + r.nextInt(2));
                break;
            case AUTUMN:
                if (isIron(state)) return ModItems.make(ModItems.Id.RUSTED_SCRAP, 2 + r.nextInt(4));
                if (isCoal(state)) return ModItems.make(ModItems.Id.AMBER_RESIN, 1 + r.nextInt(3));
                break;
            default:
                break;
        }
        return null;
    }
}

package com.planets;

import java.util.List;
import java.util.Set;
import net.minecraft.ChatFormatting;
import net.minecraft.core.BlockPos;
import net.minecraft.core.component.DataComponents;
import net.minecraft.network.chat.Component;
import net.minecraft.server.MinecraftServer;

import net.minecraft.tags.BlockTags;
import net.minecraft.util.RandomSource;
import net.minecraft.world.SimpleContainer;

import net.minecraft.world.item.ItemStack;
import net.minecraft.world.item.Items;
import net.minecraft.world.item.component.ItemLore;
import net.minecraft.world.level.levelgen.Heightmap;

public final class PlanetTravel {
    private PlanetTravel() {}

    private static final int[] PLANET_SLOTS = {10, 12, 14, 16};
    private static final int HOME_SLOT = 22;

    public static void openMenu(ServerPlayer player) {
        SimpleContainer container = new SimpleContainer(27);

        ItemStack filler = new ItemStack(Items.GRAY_STAINED_GLASS_PANE);
        filler.set(DataComponents.CUSTOM_NAME, Component.literal(" "));
        for (int i = 0; i < 27; i++) container.setItem(i, filler.copy());

        Planet[] planets = Planet.values();
        for (int i = 0; i < planets.length; i++) {
            Planet p = planets[i];
            ItemStack icon = new ItemStack(p.icon);
            icon.set(DataComponents.CUSTOM_NAME, ModItems.text(p.display, p.color));
            icon.set(DataComponents.LORE, new ItemLore(List.of(
                    ModItems.text(p.tagline, ChatFormatting.GRAY),
                    ModItems.text("Size: " + p.size + " x " + p.size + " blocks", ChatFormatting.DARK_GRAY),
                    ModItems.text("Resources: " + p.resources, ChatFormatting.DARK_GRAY),
                    ModItems.text("Click to travel", ChatFormatting.GREEN))));
            container.setItem(PLANET_SLOTS[i], icon);
        }

        NationManager.Nation nation = PlanetsMod.NATIONS.byPlayer(player.getUUID());
        if (nation != null && nation.homePlanet != null) {
            ItemStack home = new ItemStack(Items.RED_BED);
            home.set(DataComponents.CUSTOM_NAME, ModItems.text("Nation home: " + nation.name, ChatFormatting.RED));
            home.set(DataComponents.LORE, new ItemLore(List.of(
                    ModItems.text("Travel to your nation's home point", ChatFormatting.GRAY),
                    ModItems.text("Click to travel", ChatFormatting.GREEN))));
            container.setItem(HOME_SLOT, home);
        }

        player.openMenu(new SimpleMenuProvider(
                (syncId, inv, p) -> new ButtonMenu(ButtonMenu.typeFor(3), syncId, inv, container, 3,
                        (sp, slot) -> onClick(sp, slot)),
                Component.literal("Choose a planet")));
    }

    private static void onClick(ServerPlayer player, int slot) {
        for (int i = 0; i < PLANET_SLOTS.length; i++) {
            if (PLANET_SLOTS[i] == slot) {
                travel(player, Planet.values()[i]);
                return;
            }
        }
        if (slot == HOME_SLOT) {
            NationManager.Nation nation = PlanetsMod.NATIONS.byPlayer(player.getUUID());
            if (nation != null && nation.homePlanet != null) {
                goHome(player, nation);
            }
        }
    }

    public static void goHome(ServerPlayer player, NationManager.Nation nation) {
        Planet planet = Planet.byId(nation.homePlanet);
        ServerLevel level = planet == null ? null : planet.level(PlanetsMod.server);
        if (level == null) {
            player.sendSystemMessage(Component.literal("Your nation's home world is not available.")
                    .withStyle(ChatFormatting.RED));
            return;
        }
        player.closeContainer();
        teleport(player, level, nation.hx, nation.hy, nation.hz);
        player.sendSystemMessage(Component.literal("Welcome home to " + nation.name + ".")
                .withStyle(ChatFormatting.GREEN));
    }

    public static void travel(ServerPlayer player, Planet target) {
        MinecraftServer server = PlanetsMod.server;
        ServerLevel dest = target.level(server);
        if (dest == null) {
            player.sendSystemMessage(Component.literal(target.display + " is not loaded. Is the planets data pack active?")
                    .withStyle(ChatFormatting.RED));
            return;
        }
        player.closeContainer();
        BlockPos spot = findSpot(dest);
        teleport(player, dest, spot.getX() + 0.5, spot.getY(), spot.getZ() + 0.5);
        player.sendSystemMessage(Component.literal("Arrived on " + target.display + ".").withStyle(target.color));
    }

    public static void teleport(ServerPlayer player, ServerLevel level, double x, double y, double z) {
        player.teleportTo(level, x, y, z, Set.of(), player.getYRot(), player.getXRot(), true);
    }

    /** Looks for dry, open ground near the middle of a planet. */
    private static BlockPos findSpot(ServerLevel level) {
        RandomSource random = RandomSource.create();
        for (int i = 0; i < 40; i++) {
            int range = 16 + i * 10;
            int x = random.nextInt(range * 2 + 1) - range;
            int z = random.nextInt(range * 2 + 1) - range;
            level.getChunk(x >> 4, z >> 4); // make sure it is loaded
            int y = level.getHeight(Heightmap.Types.MOTION_BLOCKING_NO_LEAVES, x, z);
            BlockPos below = new BlockPos(x, y - 1, z);
            if (level.getFluidState(below).isEmpty()
                    && level.getFluidState(below.above()).isEmpty()
                    && !level.getBlockState(below).is(BlockTags.LEAVES)) {
                return new BlockPos(x, y, z);
            }
        }
        level.getChunk(0, 0);
        return new BlockPos(0, level.getHeight(Heightmap.Types.MOTION_BLOCKING_NO_LEAVES, 0, 0), 0);
    }
}

package com.planets;

import net.fabricmc.api.ModInitializer;
import net.fabricmc.fabric.api.event.lifecycle.v1.ServerLifecycleEvents;
import net.fabricmc.fabric.api.event.lifecycle.v1.ServerTickEvents;
import net.fabricmc.fabric.api.networking.v1.ServerPlayConnectionEvents;
import net.minecraft.server.MinecraftServer;
import net.minecraft.server.level.ServerPlayer;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

public class PlanetsMod implements ModInitializer {
    public static final String MOD_ID = "planets";
    public static final Logger LOGGER = LoggerFactory.getLogger(MOD_ID);
    public static final NationManager NATIONS = new NationManager();

    /** Set once the server has started. */
    public static MinecraftServer server;

    private static final String COMPASS_TAG = "planets_compass_given";

    @Override
    public void onInitialize() {
        NATIONS.load();
        PlanetDecorator.load();

        ServerLifecycleEvents.SERVER_STARTED.register(s -> {
            server = s;
            PlanetBorders.apply(s);
        });

        ServerLifecycleEvents.SERVER_STOPPING.register(s -> {
            NATIONS.save();
            PlanetDecorator.save();
        });

        ServerTickEvents.END_SERVER_TICK.register(s -> {
            PlanetDecorator.tick(s);
            if (s.getTickCount() % 6000 == 0) {
                PlanetDecorator.save();
            }
        });

        ServerPlayConnectionEvents.JOIN.register((handler, sender, s) -> {
            ServerPlayer player = handler.getPlayer();
            NATIONS.remember(player);
            if (!player.getTags().contains(COMPASS_TAG)) {
                player.addTag(COMPASS_TAG);
                player.getInventory().placeItemBackInInventory(ModItems.make(ModItems.Id.PLANET_COMPASS, 1));
            }
        });

        NationCommands.register();
        BlockEvents.register();
        LOGGER.info("Planets & Nations loaded.");
    }
}
